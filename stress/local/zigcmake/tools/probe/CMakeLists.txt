# zigcmake's probe: find_package of the zig-libs CMake configs (abseil,
# GEOS, oneTBB, expat), find_library and find_path, each checked to be
# under zig-libs, then one program linked against all four.
cmake_minimum_required(VERSION 3.21)
project(zigcmake_probe C CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(absl CONFIG REQUIRED)
find_package(GEOS CONFIG REQUIRED)
find_package(TBB CONFIG REQUIRED)
find_package(expat CONFIG REQUIRED)
find_library(ZLIB_ARCHIVE NAMES z REQUIRED)
find_path(EXPAT_HEADER expat.h REQUIRED)
set(found "")
foreach(v absl_DIR GEOS_DIR TBB_DIR expat_DIR ZLIB_ARCHIVE EXPAT_HEADER)
  file(TO_CMAKE_PATH "${${v}}" p)
  message(STATUS "zigcmake: ${v} ${p}")
  if(NOT p MATCHES "/zig-libs/")
    message(FATAL_ERROR "zigcmake: ${v} is not under zig-libs: ${p}")
  endif()
  string(APPEND found "${v} ${p}\n")
endforeach()
file(WRITE "${CMAKE_BINARY_DIR}/found.txt" "${found}")
add_executable(probe probe.cpp)
target_link_libraries(probe PRIVATE absl::strings GEOS::geos_c TBB::tbb expat::expat)
